namespace = polarbear

#Give polar bear trait to polar bears born with polar bear ethnicity
polarbear.0001 = {
	hidden = yes
	trigger = {
		OR = {
			real_father = {
				culture = culture:polarbear
			}
			mother = {
				culture = culture:polarbear
			}
		}
	}
	immediate = {
		add_trait = polarbear_trait
		set_culture = polarbear
	}
}

polarbear.0003 = {
	hidden = yes
	immediate = {
		every_living_character = {
			limit = {
				culture = culture:polarbear
				NOT = {
					has_trait = polarbear_trait
				}
			}
			add_trait = polarbear_trait
		}
	}
}
