namespace = polarbear

#Give polar bear trait to polar bears born with polar bear ethnicity
polarbear.0001 = {
	hidden = yes
	trigger = {
		OR = {
			real_father = {
				culture = culture:polarbear
			}
			mother = {
				culture = culture:polarbear
			}
		}
	}
	immediate = {
		add_trait = polarbear_trait
		set_culture = culture:polarbear
		add_secret = {
			type = secret_polar_bear
		}
	}
}

#Double check your polar bears on the game starting
polarbear.0002 = {
	hidden = yes
	immediate = {
		every_living_character = {
			limit = {
				culture = culture:polarbear
				NOT = {
					has_trait = polarbear_trait
				}
			}
			add_trait = polarbear_trait
			add_secret = {
				type = secret_polar_bear
			}
		}
	}
}

#Align polar bear culture and trait for appearance
polarbear.0003 = {
	hidden = yes
	trigger = {
		OR = {
			has_trait = polarbear_trait
			culture = culture:polarbear
		}
	}
	immediate = {
		set_culture = culture:polarbear
		add_trait = polarbear_trait
		add_secret = {
			type = secret_polar_bear
		}
	}
}
